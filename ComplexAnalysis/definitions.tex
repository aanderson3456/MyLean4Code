This document provides a comprehensive list of all definitions manually formalized in the \texttt{ComplexAnalysis} project, specifically within \texttt{R2.lean} and the \texttt{Sarason/} directory. For each definition, the raw Lean code is presented first, followed by its mathematical interpretation.

\section{\texttt{ComplexAnalysis/R2.lean}}

\begin{itemize}
    \item \textbf{\texttt{sqNorm}}
\begin{lstlisting}[language=Caml]
noncomputable def sqNorm (x : $\mathbb{R} \times \mathbb{R}$) : $\mathbb{R}$ := x.1$^2$ + x.2$^2$
\end{lstlisting}
    \[ \text{sqNorm}(x) = x_1^2 + x_2^2 \]

    \item \textbf{\texttt{sqDist}}
\begin{lstlisting}[language=Caml]
noncomputable def sqDist (x y : $\mathbb{R} \times \mathbb{R}$) : $\mathbb{R}$ := sqNorm (x - y)
\end{lstlisting}
    \[ \text{sqDist}(x, y) = \text{sqNorm}(x - y) \]

    \item \textbf{\texttt{euclideanNorm}}
\begin{lstlisting}[language=Caml]
noncomputable def euclideanNorm (x : $\mathbb{R} \times \mathbb{R}$) : $\mathbb{R}$ := Real.sqrt (sqNorm x)
\end{lstlisting}
    \[ \text{euclideanNorm}(x) = \sqrt{x_1^2 + x_2^2} \]

    \item \textbf{\texttt{euclideanDist}}
\begin{lstlisting}[language=Caml]
noncomputable def euclideanDist (x y : $\mathbb{R} \times \mathbb{R}$) : $\mathbb{R}$ := Real.sqrt (sqDist x y)
\end{lstlisting}
    \[ \text{euclideanDist}(x, y) = \sqrt{\text{sqDist}(x, y)} \]

    \item \textbf{\texttt{LimitRtoR2}}
\begin{lstlisting}[language=Caml]
def LimitRtoR2 (f : $\mathbb{R} \to \mathbb{R} \times \mathbb{R}$) (a : $\mathbb{R}$) (L : $\mathbb{R} \times \mathbb{R}$) : Prop := 
  $\forall\ \varepsilon > 0,\ \exists\ \delta > 0,\ \forall$ x : $\mathbb{R}$, 0 < |x - a| $\wedge$ |x - a| < $\delta \to$ euclideanDist (f x) L < $\varepsilon$
\end{lstlisting}
    \[ \lim_{x \to a} f(x) = L \iff \forall \epsilon > 0, \exists \delta > 0, \forall x, 0 < |x - a| < \delta \implies \text{euclideanDist}(f(x), L) < \epsilon \]

    \item \textbf{\texttt{LimitR2toR}}
\begin{lstlisting}[language=Caml]
def LimitR2toR (f : $\mathbb{R} \times \mathbb{R} \to \mathbb{R}$) (a : $\mathbb{R} \times \mathbb{R}$) (L : $\mathbb{R}$) : Prop := 
  $\forall\ \varepsilon > 0,\ \exists\ \delta > 0,\ \forall$ x : $\mathbb{R} \times \mathbb{R}$, 0 < euclideanDist x a $\wedge$ euclideanDist x a < $\delta \to$ |f x - L| < $\varepsilon$
\end{lstlisting}
    \[ \lim_{x \to a} f(x) = L \iff \forall \epsilon > 0, \exists \delta > 0, \forall x \in \mathbb{R}^2, 0 < \text{euclideanDist}(x, a) < \delta \implies |f(x) - L| < \epsilon \]

    \item \textbf{\texttt{LimitR2toR2}}
\begin{lstlisting}[language=Caml]
def LimitR2toR2 (f : $\mathbb{R} \times \mathbb{R} \to \mathbb{R} \times \mathbb{R}$) (a : $\mathbb{R} \times \mathbb{R}$) (L : $\mathbb{R} \times \mathbb{R}$) : Prop := 
  $\forall\ \varepsilon > 0,\ \exists\ \delta > 0,\ \forall$ x : $\mathbb{R} \times \mathbb{R}$, 0 < euclideanDist x a $\wedge$ euclideanDist x a < $\delta \to$ euclideanDist (f x) L < $\varepsilon$
\end{lstlisting}
    \[ \lim_{x \to a} f(x) = L \iff \forall \epsilon > 0, \exists \delta > 0, \forall x \in \mathbb{R}^2, 0 < \text{euclideanDist}(x, a) < \delta \implies \text{euclideanDist}(f(x), L) < \epsilon \]

    \item \textbf{\texttt{ConvergesR2}}
\begin{lstlisting}[language=Caml]
def ConvergesR2 (seq : $\mathbb{N} \to \mathbb{R} \times \mathbb{R}$) (L : $\mathbb{R} \times \mathbb{R}$): Prop := 
  $\forall\ \varepsilon > 0,\ \exists$ N : $\mathbb{N},\ \forall$ n $\ge$ N, euclideanDist (seq n) L < $\varepsilon$
\end{lstlisting}
    \[ \lim_{n \to \infty} \text{seq}(n) = L \iff \forall \epsilon > 0, \exists N \in \mathbb{N}, \forall n \ge N, \text{euclideanDist}(\text{seq}(n), L) < \epsilon \]

    \item \textbf{\texttt{HasFDerivAt\_R2\_eps}}
\begin{lstlisting}[language=Caml]
def HasFDerivAt_R2_eps (u v : $\mathbb{R} \times \mathbb{R} \to \mathbb{R}$) (ux0 uy0 vx0 vy0 : $\mathbb{R}$) (a : $\mathbb{R} \times \mathbb{R}$) : Prop :=
  LimitR2toR2 (fun h => (
    (u (a.1 + h.1, a.2 + h.2) - u a) - (ux0 * h.1 + uy0 * h.2), 
    (v (a.1 + h.1, a.2 + h.2) - v a) - (vx0 * h.1 + vy0 * h.2)
  )) (0, 0) (0, 0)
\end{lstlisting}
    \[ \lim_{h \to 0} \left( F(a+h) - F(a) - \begin{pmatrix} u_{x0} & u_{y0} \\ v_{x0} & v_{y0} \end{pmatrix} h \right) = 0 \]

    \item \textbf{\texttt{DifferentiableAt\_R2\_eps}}
\begin{lstlisting}[language=Caml]
def DifferentiableAt_R2_eps (u v : $\mathbb{R} \times \mathbb{R} \to \mathbb{R}$) (a : $\mathbb{R} \times \mathbb{R}$) : Prop := 
  $\exists$ ux0 uy0 vx0 vy0 : $\mathbb{R}$, HasFDerivAt_R2_eps u v ux0 uy0 vx0 vy0 a
\end{lstlisting}
    \[ \exists u_{x0}, u_{y0}, v_{x0}, v_{y0} \in \mathbb{R}, \text{HasFDerivAt\_R2\_eps}(u, v, u_{x0}, u_{y0}, v_{x0}, v_{y0}, a) \]

    \item \textbf{\texttt{HasDerivAt\_RtoR2\_eps}}
\begin{lstlisting}[language=Caml]
def HasDerivAt_RtoR2_eps (f : $\mathbb{R} \to \mathbb{R} \times \mathbb{R}$) (f'0 : $\mathbb{R} \times \mathbb{R}$) (t$_0$ : $\mathbb{R}$) : Prop :=
  LimitRtoR2 (fun t => (
    (f t).1 - (f t$_0$).1 - (t - t$_0$) * f'0.1, 
    (f t).2 - (f t$_0$).2 - (t - t$_0$) * f'0.2
  )) t$_0$ (0, 0)
\end{lstlisting}
    \[ \lim_{t \to t_0} (f(t) - f(t_0) - (t - t_0)f'_0) = 0 \]
\end{itemize}

\section{\texttt{ComplexAnalysis/Sarason/Definitions.lean}}

\begin{itemize}
    \item \textbf{\texttt{HasLimitAt\_eps}}
\begin{lstlisting}[language=Caml]
def HasLimitAt_eps (f : $\mathbb{C} \to \mathbb{C}$) (L : $\mathbb{C}$) (z$_0$ : $\mathbb{C}$) : Prop := 
  $\forall\ \varepsilon > 0,\ \exists\ \delta > 0,\ \forall$ z : $\mathbb{C}$, 0 < Complex.abs (z - z$_0$) $\wedge$ Complex.abs (z - z$_0$) < $\delta \to$ Complex.abs (f z - L) < $\varepsilon$
\end{lstlisting}
    \[ \lim_{z \to z_0} f(z) = L \]

    \item \textbf{\texttt{HasLimitAtInf\_eps}}
\begin{lstlisting}[language=Caml]
def HasLimitAtInf_eps (f : $\mathbb{C} \to \mathbb{R}$) (L : $\mathbb{R}$) : Prop := 
  $\forall\ \varepsilon > 0,\ \exists$ R > 0, $\forall$ z : $\mathbb{C}$, Complex.abs z > R $\to$ |f z - L| < $\varepsilon$
\end{lstlisting}
    \[ \lim_{|z| \to \infty} f(z) = L \]

    \item \textbf{\texttt{HasDerivAt\_eps}}
\begin{lstlisting}[language=Caml]
def HasDerivAt_eps (f : $\mathbb{C} \to \mathbb{C}$) (f' : $\mathbb{C}$) (z$_0$ : $\mathbb{C}$) : Prop := 
  HasLimitAt_eps (fun z => (f z - f z$_0$) / (z - z$_0$)) f' z$_0$
\end{lstlisting}
    \[ \lim_{z \to z_0} \frac{f(z) - f(z_0)}{z - z_0} = f' \]

    \item \textbf{\texttt{DifferentiableAt\_eps}}
\begin{lstlisting}[language=Caml]
def DifferentiableAt_eps (f : $\mathbb{C} \to \mathbb{C}$) (z$_0$ : $\mathbb{C}$) : Prop := 
  $\exists$ f', HasDerivAt_eps f f' z$_0$
\end{lstlisting}
    \[ f \text{ is differentiable at } z_0 \]

    \item \textbf{\texttt{HasPartialDerivX\_C\_to\_R\_eps}}
\begin{lstlisting}[language=Caml]
def HasPartialDerivX_C_to_R_eps (u : $\mathbb{C} \to \mathbb{R}$) (ux : $\mathbb{R}$) (z$_0$ : $\mathbb{C}$) : Prop := 
  $\forall\ \varepsilon > 0,\ \exists\ \delta > 0,\ \forall$ x : $\mathbb{R}$, 0 < |x - z$_0$.re| $\wedge$ |x - z$_0$.re| < $\delta \to$ 
    |(u ((x : $\mathbb{C}$) + (z$_0$.im : $\mathbb{C}$) * I) - u z$_0$) / (x - z$_0$.re) - ux| < $\varepsilon$
\end{lstlisting}
    \[ \frac{\partial u}{\partial x}(z_0) = u_x \]

    \item \textbf{\texttt{HasPartialDerivY\_C\_to\_R\_eps}}
\begin{lstlisting}[language=Caml]
def HasPartialDerivY_C_to_R_eps (u : $\mathbb{C} \to \mathbb{R}$) (uy : $\mathbb{R}$) (z$_0$ : $\mathbb{C}$) : Prop := 
  $\forall\ \varepsilon > 0,\ \exists\ \delta > 0,\ \forall$ y : $\mathbb{R}$, 0 < |y - z$_0$.im| $\wedge$ |y - z$_0$.im| < $\delta \to$ 
    |(u ((z$_0$.re : $\mathbb{C}$) + (y : $\mathbb{C}$) * I) - u z$_0$) / (y - z$_0$.im) - uy| < $\varepsilon$

\begin{lstlisting}[language=Caml]
noncomputable def deriv_eps (f : $\mathbb{C} \to \mathbb{C}$) (z$_0$ : $\mathbb{C}$) (h : DifferentiableAt_eps f z$_0$) : $\mathbb{C}$ :=

    \item \textbf{\texttt{HasDerivAt\_R\_to\_C\_eps}}
\begin{lstlisting}[language=Caml]
def HasDerivAt_R_to_C_eps (f : $\mathbb{R} \to \mathbb{C}$) (f'0 : $\mathbb{C}$) (t$_0$ : $\mathbb{R}$) : Prop := 
  $\forall\ \varepsilon > 0,\ \exists\ \delta > 0,\ \forall$ t : $\mathbb{R}$, 0 < |t - t$_0$| $\wedge$ |t - t$_0$| < $\delta \to$

\end{lstlisting}
    \[ \lim_{t \to t_0} \frac{f(t) - f(t_0)}{t - t_0} = f'_0 \]

    \item \textbf{\texttt{HolomorphicAt\_eps}}
\begin{lstlisting}[language=Caml]
def HolomorphicAt_eps (f : $\mathbb{C} \to \mathbb{C}$) (z$_0$ : $\mathbb{C}$) : Prop := 
  $\exists$ r > 0, $\forall$ z, Complex.abs (z - z$_0$) < r $\to$ DifferentiableAt_eps f z
\end{lstlisting}
    \[ f \text{ is holomorphic at } z_0 \iff f \text{ is complex-differentiable on some open ball around } z_0 \]

\end{lstlisting}
    \[ f \text{ is holomorphic on } G \]

noncomputable def $\Phi$ (z : $\mathbb{C}$) : EuclideanSpace $\mathbb{R}$ (Fin 3) :=

\begin{lstlisting}[language=Caml]
def del (f : $\mathbb{C} \to \mathbb{C}$) (z : $\mathbb{C}$) : $\mathbb{C}$ := 
  (1/2 : $\mathbb{C}$) * (deriv (fun x : $\mathbb{R} \Rightarrow$ f (z + x)) 0 - I * deriv (fun y : $\mathbb{R} \Rightarrow$ f (z + y * I)) 0)
\end{lstlisting}

def conformal (f : $\mathbb{C} \to \mathbb{C}$) (z$_0$ : $\mathbb{C}$) : Prop := 
  $\forall$ x$_1$ y$_1$ x$_2$ y$_2$ : $\mathbb{C},\ \forall\ \gamma_1$ : path_in_C x$_1$ y$_1,\ \forall$ t$_1$ : $\mathbb{R}$, 
  $\forall\ \gamma_2$ : path_in_C x$_2$ y$_2,\ \forall$ t$_2$ : $\mathbb{R}$,
  $\gamma_1$.extend t$_1$ = z$_0 \to \gamma_2$.extend t$_2$ = z$_0 \to$ 
  DifferentiableAt $\mathbb{R}\ \gamma_1$.extend t$_1 \to$ DifferentiableAt $\mathbb{R}\ \gamma_2$.extend t$_2 \to$

  pathDeriv (f_circ_gamma $\gamma_1$ f) t$_1 \neq$ 0 $\wedge$ 
  pathDeriv (f_circ_gamma $\gamma_2$ f) t$_2 \neq$ 0 $\wedge$ 
  pathAngle (f_circ_gamma $\gamma_1$ f) t$_1$ (f_circ_gamma $\gamma_2$ f) t$_2$ = pathAngle $\gamma_1$ t$_1\ \gamma_2$ t$_2$
\end{lstlisting}
